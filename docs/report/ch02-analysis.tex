\chapter[تحلیل نیازمندی‌ها]{تحلیل نیازمندی‌ها و مدل کسب‌وکار}

\section{بازیگران سیستم}

دو بازیگر انسانی اصلی وجود دارد. نقش در زمان ثبت‌نام انتخاب می‌شود و سرور آن را با پرچم‌های \en{is\_driver} و \en{is\_operator} نگه می‌دارد. کلاینت هنگام ورود بررسی می‌کند که نقش انتخاب‌شده در صفحهٔ ورود با نقش ذخیره‌شده در حساب یکی باشد؛ در غیر این صورت ورود رد می‌شود.

\begin{description}
  \item[راننده:] صاحب یا بهره‌بردار یک دستگاه حمل با پلاک ایرانی، نوع بارگیر (ماشین)، مدل و استان فعالیت. او بار باز می‌بیند، می‌پذیرد، مسیریابی می‌کند و تحویل را تأیید می‌کند.
  \item[متصدی حمل‌ونقل (اپراتور):] ثبت‌کننده و مدیر بار. بار ایجاد می‌کند، جزئیات را ویرایش می‌کند، در صورت نیاز حذف می‌کند و پس از انجام کار، تکمیل را از سمت خود تأیید می‌کند.
\end{description}

یک کاربر نمی‌تواند همزمان از یک حساب واحد هر دو نقش را در کلاینت فعال کند؛ تمایز نقش در قرارداد \textenglish{API} صریح است (\en{user\_type} برابر \en{driver} یا \en{operator}).

\section{چرخهٔ عمر بار}

بار در سرور با فیلد وضعیت نگهداری می‌شود. مقادیر خام دامنه به زبان انگلیسی هستند و کلاینت آن‌ها را به برچسب فارسی نگاشت می‌کند (جدول~\ref{tab:status}).

\begin{itemize}

\section{نیازمندی‌های کارکردی}

\subsection{حساب کاربری}
ورود با شماره موبایل ایران و رمز عبور؛ ثبت‌نام با نام، نام خانوادگی، تأیید رمز، و برای راننده انتخاب نوع ماشین و استان؛ مشاهده و ویرایش پروفایل؛ تغییر رمز با رمز قدیم؛ خروج و ابطال نشست محلی.

\subsection{مدیریت بار (متصدی)}
ایجاد بار با عنوان، توضیحات، کالا (\en{product})، نوع ماشین (\en{machine})، وزن، قیمت، استان و نشانی مبدأ و مقصد؛ فهرست بارهای متعلق به متصدی احراز هویت‌شده؛ جزئیات بار و رانندهٔ تخصیص‌یافته در صورت وجود؛ ویرایش و حذف؛ تأیید تکمیل.

\subsection{عملیات راننده}
فهرست صفحه‌بندی‌شدهٔ بارهای باز با فیلتر استان مبدأ، استان مقصد و بازهٔ قیمت؛ پیشنهاد بار نزدیک در شعاع ۲۵ کیلومتر بر اساس \textenglish{GPS} واقعی؛ اجبار تکمیل اطلاعات خودرو قبل از پذیرش؛ پذیرش بار؛ فهرست مأموریت‌های تخصیص‌یافته؛ مسیریابی تا مبدأ و مقصد؛ تأیید تکمیل از سمت راننده.

\subsection{دادهٔ مرجع}
فهرست کالاها، انواع ماشین و استان‌ها از سرور خوانده می‌شود تا شناسه‌های خارجی در بار و پروفایل راننده یکسان بماند.

\section{نیازمندی‌های غیرکارکردی}

\begin{itemize}
  \item \textbf{امنیت:} ژتون دسترسی در ذخیره‌ساز امن دستگاه؛ ارسال در سرآیند \en{Authorization: Bearer}؛ مهلت بی‌فعالیتی ۱۵ دقیقه برای اجبار ورود مجدد.
  \item \textbf{بومی‌سازی:} فارسی پیش‌فرض با چینش راست‌به‌چپ و فونت وزیر؛ انگلیسی با فونت \textenglish{Inter}؛ تقویم و ارقام در متن‌های کاربری فارسی.
  \item \textbf{دسترسی‌پذیری ظاهری:} حالت روشن و تاریک با رنگ سازمانی سبز تهرانی (\textenglish{\#0F766E}).
  \item \textbf{چندسکویی:} یک کدبیس فلاتر برای اندروید و وب؛ نسخهٔ \textenglish{PWA} قابل نصب با کارکرد آفلاین محدود پوستهٔ برنامه.
  \item \textbf{آزمون‌پذیری:} قرارداد \textenglish{OpenAPI} در آزمون‌های واحد کلاینت قفل شده است تا تغییر ناخواستهٔ فیلدها مشخص شود.
\end{itemize}

\section{موجودیت‌های دامنه}

چهار موجودیت محوری سامانه عبارت‌اند از کاربر، بار (\en{Bar})، تخصیص (\en{Assign}) و داده‌های مرجع (کالا، ماشین، استان). بار به کالا و ماشین و دو استان مبدأ/مقصد وصل است. تخصیص، بار را به راننده پیوند می‌دهد و دو پرچم تأیید را نگه می‌دارد. جزئیات فیلدها در فصل پایگاه داده آمده است.

\section{سناریوی اصلی موفقیت}

\begin{enumerate}
  \item متصدی وارد می‌شود و بار «حمل سیمان تهران به اصفهان» را با ماشین کفی و وزن مشخص ثبت می‌کند. وضعیت \en{open} است.
  \item رانندهٔ کفی وارد می‌شود، در صورت نیاز پلاک و مدل را کامل می‌کند، بار را در فهرست یا در بارهای نزدیک می‌بیند و می‌پذیرد.
  \item سرور تخصیص می‌سازد و وضعیت \en{assign} می‌شود. راننده مأموریت را در صفحهٔ مأموریت‌ها می‌بیند.
  \item راننده مسیر را تا مبدأ بار و سپس مقصد روی نقشهٔ نشان دنبال می‌کند.
  \item پس از تخلیه، راننده تکمیل را می‌زند (\en{confirm\_driver}). متصدی نیز تکمیل را تأیید می‌کند (\en{confirm\_operator}). بار \en{done} می‌شود.
\end{enumerate}
